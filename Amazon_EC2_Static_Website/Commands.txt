cat > Commands.txt <<'EOF'
# ============================================================
# Amazon EC2 Static Website - Commands
# ============================================================

# ------------------------------------------------------------
# 1. AWS CLI / Account Configuration
# ------------------------------------------------------------

aws --version

aws configure set region ap-south-1 --profile new-aws

aws sts get-caller-identity --profile new-aws

aws configure list --profile new-aws

aws configure get region --profile new-aws

# Disable AWS CLI pager
$env:AWS_PAGER=""

# ------------------------------------------------------------
# 2. AWS Network Information
# ------------------------------------------------------------

$VPC_ID = aws ec2 describe-vpcs --filters "Name=is-default,Values=true" --query "Vpcs[0].VpcId" --output text --profile new-aws --region ap-south-1

$SUBNET_ID = aws ec2 describe-subnets --filters "Name=vpc-id,Values=$VPC_ID" --query "Subnets[0].SubnetId" --output text --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 3. EC2 Key Pair
# ------------------------------------------------------------

aws ec2 create-key-pair --key-name Amazon-EC2-Static-Website-Key --query "KeyMaterial" --output text --profile new-aws --region ap-south-1 | Out-File -Encoding ascii Amazon-EC2-Static-Website-Key.pem

# ------------------------------------------------------------
# 4. Security Group
# ------------------------------------------------------------

$MY_IP = (Invoke-RestMethod https://checkip.amazonaws.com).Trim()

$SG_ID = aws ec2 create-security-group --group-name Amazon-EC2-Static-Website-SG --description "Security group for EC2 static website" --vpc-id $VPC_ID --query "GroupId" --output text --profile new-aws --region ap-south-1

aws ec2 authorize-security-group-ingress --group-id $SG_ID --protocol tcp --port 22 --cidr "$MY_IP/32" --profile new-aws --region ap-south-1

aws ec2 authorize-security-group-ingress --group-id $SG_ID --protocol tcp --port 80 --cidr 0.0.0.0/0 --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 5. Obtain Ubuntu 22.04 AMI
# ------------------------------------------------------------

$AMI_ID = aws ec2 describe-images --owners 099720109477 --filters "Name=name,Values=ubuntu/images/hvm-ssd/ubuntu-jammy-22.04-amd64-server-*" "Name=state,Values=available" "Name=architecture,Values=x86_64" "Name=root-device-type,Values=ebs" --query "sort_by(Images,&CreationDate)[-1].ImageId" --output text --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 6. Launch EC2 Instance
# Note: t3.micro used because t2.micro was not eligible for
# the Free Tier for this newly created AWS account.
# ------------------------------------------------------------

$INSTANCE_ID = aws ec2 run-instances --image-id $AMI_ID --instance-type t3.micro --key-name Amazon-EC2-Static-Website-Key --security-group-ids $SG_ID --subnet-id $SUBNET_ID --associate-public-ip-address --tag-specifications 'ResourceType=instance,Tags=[{Key=Name,Value=Amazon-EC2-Static-Website}]' --query "Instances[0].InstanceId" --output text --profile new-aws --region ap-south-1

aws ec2 wait instance-running --instance-ids $INSTANCE_ID --profile new-aws --region ap-south-1

$PUBLIC_IP = aws ec2 describe-instances --instance-ids $INSTANCE_ID --query "Reservations[0].Instances[0].PublicIpAddress" --output text --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 7. Verify EC2 Status
# ------------------------------------------------------------

aws ec2 describe-instance-status --instance-ids $INSTANCE_ID --include-all-instances --query "InstanceStatuses[0].[InstanceState.Name,SystemStatus.Status,InstanceStatus.Status]" --output table --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 8. Systems Manager Setup
# Used because direct SSH connectivity was unavailable.
# ------------------------------------------------------------

$RoleName="AmazonEC2SSMRole"

$TrustPolicy='{"Version":"2012-10-17","Statement":[{"Effect":"Allow","Principal":{"Service":"ec2.amazonaws.com"},"Action":"sts:AssumeRole"}]}'

aws iam create-role --role-name $RoleName --assume-role-policy-document $TrustPolicy --profile new-aws

aws iam attach-role-policy --role-name $RoleName --policy-arn arn:aws:iam::aws:policy/AmazonSSMManagedInstanceCore --profile new-aws

$ProfileName="AmazonEC2SSMInstanceProfile"

aws iam create-instance-profile --instance-profile-name $ProfileName --profile new-aws

aws iam add-role-to-instance-profile --instance-profile-name $ProfileName --role-name $RoleName --profile new-aws

Start-Sleep -Seconds 15

aws ec2 associate-iam-instance-profile --instance-id $INSTANCE_ID --iam-instance-profile Name=$ProfileName --profile new-aws --region ap-south-1

aws ec2 reboot-instances --instance-ids $INSTANCE_ID --profile new-aws --region ap-south-1

aws ec2 wait instance-status-ok --instance-ids $INSTANCE_ID --profile new-aws --region ap-south-1

aws ssm describe-instance-information --filters "Key=InstanceIds,Values=$INSTANCE_ID" --query "InstanceInformationList[0].[InstanceId,PingStatus,PlatformName,AgentVersion]" --output table --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 9. Update Ubuntu + Install Apache
# ------------------------------------------------------------

$COMMAND_ID = aws ssm send-command --instance-ids $INSTANCE_ID --document-name "AWS-RunShellScript" --parameters 'commands=["sudo apt update","sudo apt upgrade -y","sudo apt install apache2 -y","sudo systemctl start apache2","sudo systemctl enable apache2","sudo systemctl status apache2 --no-pager"]' --query "Command.CommandId" --output text --profile new-aws --region ap-south-1

Start-Sleep -Seconds 15

aws ssm get-command-invocation --command-id $COMMAND_ID --instance-id $INSTANCE_ID --query "[Status,StandardOutputContent,StandardErrorContent]" --output text --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 10. Website Directory
# ------------------------------------------------------------

# Apache website directory:
# /var/www/html

# Remove default page
$COMMAND_ID = aws ssm send-command --instance-ids $INSTANCE_ID --document-name "AWS-RunShellScript" --parameters 'commands=["sudo rm -f /var/www/html/index.html"]' --query "Command.CommandId" --output text --profile new-aws --region ap-south-1

Start-Sleep -Seconds 5

aws ssm get-command-invocation --command-id $COMMAND_ID --instance-id $INSTANCE_ID --query "[Status,StandardOutputContent,StandardErrorContent]" --output text --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 11. Create Static Website
# ------------------------------------------------------------

# Website content:
#
# <!DOCTYPE html>
# <html>
# <head>
#     <title>My AWS Website</title>
# </head>
# <body>
#     <h1>Hello AWS!</h1>
#     <p>Website hosted on Amazon EC2.</p>
# </body>
# </html>

$COMMAND_ID = aws ssm send-command --instance-ids $INSTANCE_ID --document-name "AWS-RunShellScript" --parameters 'commands=["printf ''<!DOCTYPE html>\n<html>\n<head>\n<title>My AWS Website</title>\n</head>\n<body>\n<h1>Hello AWS!</h1>\n<p>Website hosted on Amazon EC2.</p>\n</body>\n</html>\n'' | sudo tee /var/www/html/index.html > /dev/null"]' --query "Command.CommandId" --output text --profile new-aws --region ap-south-1

Start-Sleep -Seconds 5

aws ssm get-command-invocation --command-id $COMMAND_ID --instance-id $INSTANCE_ID --query "[Status,StandardOutputContent,StandardErrorContent]" --output text --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 12. Set Permissions and Restart Apache
# ------------------------------------------------------------

$COMMAND_ID = aws ssm send-command --instance-ids $INSTANCE_ID --document-name "AWS-RunShellScript" --parameters 'commands=["sudo chown -R www-data:www-data /var/www/html","sudo chmod -R 755 /var/www/html","sudo systemctl restart apache2","sudo systemctl status apache2 --no-pager"]' --query "Command.CommandId" --output text --profile new-aws --region ap-south-1

Start-Sleep -Seconds 5

aws ssm get-command-invocation --command-id $COMMAND_ID --instance-id $INSTANCE_ID --query "[Status,StandardOutputContent,StandardErrorContent]" --output text --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 13. Final Website Verification
# ------------------------------------------------------------

Invoke-WebRequest "http://$PUBLIC_IP" -UseBasicParsing

# Expected:
# StatusCode : 200

# Browser:
# http://<PUBLIC-IP>

# Expected page:
# Hello AWS!
# Website hosted on Amazon EC2.

# ------------------------------------------------------------
# 14. Final Apache and Website Proof
# ------------------------------------------------------------

$COMMAND_ID = aws ssm send-command --instance-ids $INSTANCE_ID --document-name "AWS-RunShellScript" --parameters 'commands=["sudo systemctl is-active apache2","ls -l /var/www/html","sudo cat /var/www/html/index.html"]' --query "Command.CommandId" --output text --profile new-aws --region ap-south-1

Start-Sleep -Seconds 5

aws ssm get-command-invocation --command-id $COMMAND_ID --instance-id $INSTANCE_ID --query "[Status,StandardOutputContent,StandardErrorContent]" --output text --profile new-aws --region ap-south-1

# ------------------------------------------------------------
# 15. Final EC2 Verification
# ------------------------------------------------------------

aws ec2 describe-instances --instance-ids $INSTANCE_ID --query "Reservations[0].Instances[0].[InstanceId,InstanceType,State.Name,PublicIpAddress]" --output table --profile new-aws --region ap-south-1

aws ssm describe-instance-information --filters "Key=InstanceIds,Values=$INSTANCE_ID" --query "InstanceInformationList[0].[InstanceId,PingStatus,PlatformName]" --output table --profile new-aws --region ap-south-1
EOF
