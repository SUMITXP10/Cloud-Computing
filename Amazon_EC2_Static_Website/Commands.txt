    --output text `

9. UBUNTU 22.04 AMI DISCOVERY
============================================================

$AMI_ID = aws ssm get-parameter `
    --name "/aws/service/canonical/ubuntu/server/22.04/stable/current/amd64/hvm/ebs-gp3/ami-id" `

    --output text `
    --profile new-aws `

============================================================
10. LAUNCH EC2 INSTANCE
============================================================

$INSTANCE_ID = aws ec2 run-instances `
    --image-id $AMI_ID `
    --instance-type t3.micro `
    --key-name Amazon-EC2-Static-Website-Key `
    --security-group-ids $SG_ID `
    --subnet-id $SUBNET_ID `
    --associate-public-ip-address `
    --tag-specifications 'ResourceType=instance,Tags=[{Key=Name,Value=Amazon-EC2-Static-Website}]' `
    --query "Instances[0].InstanceId" `
    --output text `
    --profile new-aws `
    --region ap-south-1

$INSTANCE_ID

aws ec2 wait instance-running `
    --instance-ids $INSTANCE_ID `
    --profile new-aws `
    --region ap-south-1

17. INSTALL APACHE
============================================================

$COMMAND_ID = aws ssm send-command `
    --instance-ids $INSTANCE_ID `
    --document-name "AWS-RunShellScript" `
    --parameters 'commands=["sudo apt update","sudo apt upgrade -y","sudo apt install apache2 -y","sudo systemctl start apache2","sudo systemctl enable apache2","sudo systemctl status apache2 --no-pager"]' `
    --query "Command.CommandId" `
    --output text `
    --profile new-aws `
    --region ap-south-1

Start-Sleep -Seconds 15

aws ssm get-command-invocation `
    --command-id $COMMAND_ID `
    --instance-id $INSTANCE_ID `
    --query "[Status,StandardOutputContent,StandardErrorContent]" `
    --output text `
    --profile new-aws `
    --region ap-south-1

19. PREPARE LOCAL HTML FOR DEPLOYMENT
============================================================

$HTML_B64 = [Convert]::ToBase64String(
    [System.IO.File]::ReadAllBytes(
        (Join-Path (Get-Location) "index.html")
    )
)

$HTML_B64.Length


============================================================
20. DEPLOY INDEX.HTML
============================================================

$SSM_COMMANDS = @(
    "echo '$HTML_B64' | base64 -d | sudo tee /var/www/html/index.html > /dev/null",
    "sudo chown -R www-data:www-data /var/www/html",
    "sudo chmod -R 755 /var/www/html",
    "sudo systemctl restart apache2",
    "sudo systemctl is-active apache2"
)

$PARAMETERS = @{ commands = $SSM_COMMANDS } | ConvertTo-Json -Compress

$COMMAND_ID = aws ssm send-command `
    --instance-ids $INSTANCE_ID `
    --document-name "AWS-RunShellScript" `
    --parameters $PARAMETERS `
    --query "Command.CommandId" `
    --output text `
    --profile new-aws `
    --region ap-south-1

$COMMAND_ID

Start-Sleep -Seconds 10

aws ssm get-command-invocation `
    --command-id $COMMAND_ID `
    --instance-id $INSTANCE_ID `
    --query "[Status,StatusDetails,StandardOutputContent,StandardErrorContent]" `
    --output text `
    --profile new-aws `
    --region ap-south-1

30. DEPLOY FINAL WEBSITE TO /var/www/html

    "sudo curl -fL '$HTML_URL' -o /var/www/html/index.html",
    "sudo curl -fL '$IMAGE_URL' -o /var/www/html/My_Day.png",
    "sudo chown www-data:www-data /var/www/html/index.html /var/www/html/My_Day.png",
    "sudo chmod 755 /var/www/html/index.html",
    "sudo chmod 644 /var/www/html/My_Day.png",
    "sudo systemctl restart apache2",
    "sudo systemctl is-active apache2"
)

    --query "Command.CommandId" `
    --output text `
    --profile new-aws `
    --region ap-south-1
